### Passo 3: Implementar parser de ZIP (se Opção A)

Se não existe parser de ZIP no repo, criar `scripts/extract_zip_members.py`:

```python
import zipfile, json, hashlib
from pathlib import Path

def extract_and_hash(zip_path, output_dir):
    results = []
    with zipfile.ZipFile(zip_path) as zf:
        for member in zf.infolist():
            if member.is_dir():
                continue
            data = zf.read(member.filename)
            sha = hashlib.sha256(data).hexdigest()
            out_path = output_dir / f"{sha}_{member.filename}"
            out_path.write_bytes(data)
            results.append({
                "original_name": member.filename,
                "sha256": sha,
                "size": member.file_size,
                "stored_path": str(out_path)
            })
    return results
```
